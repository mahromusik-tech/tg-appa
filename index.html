Симулятор майнинг‑бизнеса — готовый HTML+JS (вертикально, TG Mini App ready)

Ниже полный самодостаточный файл index.html. Вертикальная страница, мобильный UI, сохранение в localStorage. Начальный баланс $1000 и старый компьютер с одной слабой видеокартой, которая даёт ~$1/с. В магазине — покупка слотов (ячейки), видеокарт разных классов, улучшения электроснабжения, охлаждения, обслуживание и временные баффы. Нельзя купить карту без свободной ячейки; при продаже карта возвращает 50% от стоимости. Электроэнергии (мощности) хватает на ограничённое количество карт — нужно апгрейдить питание.

Копируйте целиком в index.html и запускайте.

<script>
(() => {
  // --- Game data & defaults ---
  const SAVE_KEY = 'miner_tycoon_v1';
  const startMoney = 1000;

  const shopGPUs = [
    { id:'starter', name:'Старая видеокарта', price:100, hashrate:10, power:75, reliability:0.998, repairFactor:0.2, desc:'Слабая, даёт ~1$/с (стартовый)'},
    { id:'gpu1', name:'GTX 1060', price:300, hashrate:40, power:120, reliability:0.995, repairFactor:0.18, desc:'Бюджетная карта' },
    { id:'gpu2', name:'RTX 2060', price:900, hashrate:130, power:160, reliability:0.992, repairFactor:0.16, desc:'Средний класс' },
    { id:'gpu3', name:'RTX 3080', price:3200, hashrate:620, power:320, reliability:0.989, repairFactor:0.14, desc:'Топовая карта' },
    { id:'asic', name:'ASIC Miner X', price:9000, hashrate:3500, power:1500, reliability:0.995, repairFactor:0.10, desc:'Специализированный майнер' }
  ];

  // improvements
  const upgrades = {
    slotBasePrice: 200,
    powerUpgradeCost: 500, // +500W
    coolingCost: 350,
    maintenanceCost: 500,
    boostCost: 150
  };

  // Conversion: H/s -> $/s base. We want starter(gpu starter) hashrate 10 -> ~$1/s => rate = 0.1 $ per H/s
  const $perH = 0.1;

  // --- State ---
  let state = {
    money: startMoney,
    slotsTotal: 1,
    slots: [], // each slot: { id, gpu: nullable, overclock:bool, brokenUntil: timestamp|null }
    powerCapacity: 300, // Watts
    boostActiveUntil: 0,
    coolingLevel: 0, // reduces failure chance
    maintenanceTeam: false,
    timestamp: Date.now()
  };

  // init starter rig: old computer with starter gpu in slot 0
  function initStarter() {
    state.slotsTotal = 1;
    state.slots = [{ id: genId(), gpu: cloneGpu('starter'), overclock:false, brokenUntil:null }];
  }

  // --- Utils ---
  function now() { return Date.now(); }
  function genId(){ return Math.random().toString(36).slice(2,9); }
  function cloneGpu(id){
    const g = shopGPUs.find(x=>x.id===id);
    return g ? JSON.parse(JSON.stringify(g)) : null;
  }

  // load/save
  function save(){
    state.timestamp = now();
    localStorage.setItem(SAVE_KEY, JSON.stringify(state));
    log('Игра сохранена');
  }
  function load(){
    const raw = localStorage.getItem(SAVE_KEY);
    if(raw){
      try{
        const s = JSON.parse(raw);
        state = s;
        log('Сохранение загружено');
        return;
      }catch(e){}
    }
    // default
    initStarter();
    state.money = startMoney;
    state.powerCapacity = 300;
  }

  // --- UI refs ---
  const moneyEl = document.getElementById('money');
  const cashflowEl = document.getElementById('cashflow');
  const slotsArea = document.getElementById('slotsArea');
  const slotsCountEl = document.getElementById('slotsCount');
  const slotsUsedEl = document.getElementById('slotsUsed');
  const powerEl = document.getElementById('power');
  const storeArea = document.getElementById('storeArea');
  const totalHashEl = document.getElementById('totalHash');
  const incomeEl = document.getElementById('income');
  const elecEl = document.getElementById('elecCost');
  const netEl = document.getElementById('net');
  const logEl = document.getElementById('log');
  const priceKwhEl = document.getElementById('priceKwh');

  // buttons
  const btnAddSlot = document.getElementById('btnAddSlot');
  const btnPower = document.getElementById('btnPower');
  const btnCooling = document.getElementById('buyCooling');
  const btnMaintenance = document.getElementById('buyMaintenance');
  const btnBoost = document.getElementById('buyBoost');
  const btnReset = document.getElementById('btnReset');

  // --- Game logic ---
  load();
  renderStore();
  renderSlots();
  updateUI();

  // compute totals
  function computeTotals(){
    const activeSlots = state.slots;
    let totalHash = 0;
    let totalPower = 0;
    let potentialHash = 0;
    for(const s of activeSlots){
      if(s.gpu){
        const gpu = s.gpu;
        // broken?
        const broken = s.brokenUntil && s.brokenUntil > now();
        if(!broken){
          let hr = gpu.hashrate;
          if(s.overclock) hr *= 1.25;
          if(state.boostActiveUntil > now()) hr *= 1.5;
          totalHash += hr;
        }
        // power always consumed even if broken (fans/case) at 30% if broken
        let p = gpu.power;
        if(s.overclock) p *= 1.15;
        if(s.brokenUntil && s.brokenUntil > now()) p *= 0.3;
        totalPower += p;
      }
    }
    // revenue per sec
    const income = totalHash * $perH;
    // electricity cost per sec: (watts/1000)*($/kWh)/3600
    const pricekwh = parseFloat(priceKwhEl.value) || 0.12;
    const elecPerSec = totalPower/1000 * pricekwh / 3600;
    const net = income - elecPerSec;
    return { totalHash, totalPower, income, elecPerSec, net };
  }

  // tick (every second)
  function tick(){
    // generate income
    const t = computeTotals();
    state.money += t.net;
    // passive failure checks: each functional GPU has small chance to break per minute
    for(const s of state.slots){
      if(!s.gpu) continue;
      const broken = s.brokenUntil && s.brokenUntil > now();
      if(broken) continue;
      // base failure per minute = (1 - reliability) -> convert to per-second prob
      const baseFailPerMin = 1 - s.gpu.reliability; // e.g., 0.002 => 0.2% per minute
      const failProbPerSec = baseFailPerMin / 60;
      // modifications: cooling reduces chance, maintenance team reduces chance
      let mod = 1;
      if(state.coolingLevel > 0) mod *= 0.6;
      if(state.maintenanceTeam) mod *= 0.5;
      // overclock increases risk
      if(s.overclock) mod *= 1.6;
      // apply chance
      if(Math.random() < failProbPerSec * mod){
        const downtime = 1000 * (60 * (1 + Math.random()*2)); // 1-3 minutes down
        s.brokenUntil = now() + downtime;
        log(`${s.gpu.name} в слоте вышла из строя на ${Math.round(downtime/1000)} с.`);
      }
    }

    // auto-repair if maintenance team present (fast repair)
    if(state.maintenanceTeam){
      for(const s of state.slots){
        if(s.brokenUntil && s.brokenUntil > now()){
          // reduce remaining downtime
          s.brokenUntil = Math.max(now(), s.brokenUntil - 1000*2); // repair 2 seconds faster per tick
          if(s.brokenUntil <= now()){
            s.brokenUntil = null;
            log(`${s.gpu.name} починена командой`);
          }
        }
      }
    }

    updateUI();
  }

  // UI render
  function updateUI(){
    moneyEl.textContent = `$${state.money.toFixed(2)}`;
    const totals = computeTotals();
    cashflowEl.textContent = `${totals.net >= 0 ? '+' : ''}${totals.net.toFixed(2) } $/с`;
    slotsCountEl.textContent = state.slotsTotal;
    const used = state.slots.filter(s=>s.gpu).length;
    slotsUsedEl.textContent = used + ' / ' + state.slotsTotal;
    powerEl.textContent = `${Math.round(totals.totalPower)}W / ${state.powerCapacity}W`;
    totalHashEl.textContent = `${Math.round(totals.totalHash)} H/s`;
    incomeEl.textContent = `+${totals.income.toFixed(2)} $/с`;
    elecEl.textContent = `-${totals.elecPerSec.toFixed(3)} $/с`;
    netEl.textContent = `${totals.net >= 0 ? '+' : ''}${totals.net.toFixed(3)} $/с`;

    // highlight buttons availability
    btnAddSlot.disabled = state.money < slotPrice();
    btnPower.disabled = state.money < upgrades.powerUpgradeCost;
    btnCooling.disabled = state.money < upgrades.coolingCost || state.coolingLevel >= 1;
    btnMaintenance.disabled = state.money < upgrades.maintenanceCost || state.maintenanceTeam;
    btnBoost.disabled = state.money < upgrades.boostCost || state.boostActiveUntil > now();

    // render slots
    renderSlots();
  }

  function renderSlots(){
    slotsArea.innerHTML = '';
    // ensure slots array length
    while(state.slots.length < state.slotsTotal) state.slots.push({ id: genId(), gpu:null, overclock:false, brokenUntil:null });
    for(let i=0;i<state.slots.length;i++){
      const s = state.slots[i];
      const div = document.createElement('div');
      div.className = 'slot';
      if(!s.gpu){
        div.innerHTML = `<div style="font-size:13px">Пустой слот</div><div style="display:flex;gap:6px"><button class="buy compact" data-slot="${i}">Установить</button></div>`;
        const btn = div.querySelector('button');
        btn.addEventListener('click', ()=>{ openInstallMenu(i); });
      } else {
        const broken = s.brokenUntil && s.brokenUntil > now();
        div.innerHTML = `<div style="font-weight:800">${s.gpu.name}</div>
        <div style="font-size:12px;color:var(--muted)">${s.gpu.hashrate} H/s • ${s.gpu.power}W</div>
        <div style="display:flex;gap:6px">
          <button class="compact" data-sell="${i}">Продать 50%</button>
          <button class="compact" data-oc="${i}">${s.overclock? 'OC On' : 'OC Off'}</button>
          <button class="compact" data-repair="${i}" ${!broken ? 'disabled' : ''}>Починить</button>
        </div>
        <small class="info">${broken ? 'Сломана, ремонт до: ' + Math.round((s.brokenUntil-now())/1000) + ' с' : 'Работает'}</small>`;
        // sell
        div.querySelector('[data-sell]').addEventListener('click', ()=>{ sellGpu(i); });
        div.querySelector('[data-oc]').addEventListener('click', ()=>{ toggleOC(i); });
        div.querySelector('[data-repair]').addEventListener('click', ()=>{ repairNow(i); });
      }
      slotsArea.appendChild(div);
    }
  }

  // render store
  function renderStore(){
    storeArea.innerHTML = '';
    for(const g of shopGPUs){
      // disallow starter from store (already given)
      if(g.id === 'starter') continue;
      const card = document.createElement('div');
      card.className = 'card';
      card.innerHTML = `<div>
        <div style="font-weight:800">${g.name} — $${g.price}</div>
        <small class="info">${g.hashrate} H/s • ${g.power}W • надёжность ${(g.reliability*100).toFixed(2)}%</small>
        <small>${g.desc}</small>
      </div>
      <div><button class="buy compact" data-buy="${g.id}">Купить</button></div>`;
      card.querySelector('[data-buy]').addEventListener('click', ()=>{ buyGpu(g.id); });
      storeArea.appendChild(card);
    }
  }

  // --- Actions ---
  function slotPrice(){ // price increases with number of slots owned
    return upgrades.slotBasePrice + (state.slotsTotal - 1) * 120;
  }

  btnAddSlot.addEventListener('click', ()=>{
    const price = slotPrice();
    if(state.money < price){ log('Недостаточно средств для покупки слота'); return; }
    state.money -= price;
    state.slotsTotal += 1;
    state.slots.push({ id: genId(), gpu:null, overclock:false, brokenUntil:null });
    log(`Куплен слот за $${price}`);
    updateUI();
    save();
  });

  btnPower.addEventListener('click', ()=>{
    if(state.money < upgrades.powerUpgradeCost){ log('Нет денег на апгрейд питания'); return; }
    state.money -= upgrades.powerUpgradeCost;
    state.powerCapacity += 500;
    log(`Мощность увеличена на 500W за $${upgrades.powerUpgradeCost}`);
    updateUI(); save();
  });

  btnCooling.addEventListener('click', ()=>{
    if(state.coolingLevel >= 1){ log('Охлаждение уже установлено'); return; }
    if(state.money < upgrades.coolingCost){ log('Нет денег на охлаждение'); return; }
    state.money -= upgrades.coolingCost;
    state.coolingLevel = 1;
    log(`Установлено улучшенное охлаждение за $${upgrades.coolingCost}`);
    updateUI(); save();
  });

  btnMaintenance.addEventListener('click', ()=>{
    if(state.maintenanceTeam){ log('Сервисная команда уже нанята'); return; }
    if(state.money < upgrades.maintenanceCost){ log('Нет денег на сервисную команду'); return; }
    state.money -= upgrades.maintenanceCost;
    state.maintenanceTeam = true;
    log(`Найнята сервисная команда за $${upgrades.maintenanceCost}`);
    updateUI(); save();
  });

  btnBoost.addEventListener('click', ()=>{
    if(state.money < upgrades.boostCost){ log('Нет денег на Boost'); return; }
    state.money -= upgrades.boostCost;
    state.boostActiveUntil = now() + 60_000; // 60 sec
    log('Boost активирован: +50% хеша на 60с');
    updateUI(); save();
  });

  // buy GPU: must have free slot and enough money and power capacity
  function buyGpu(gpuId){
    const freeIndex = state.slots.findIndex(s=>!s.gpu);
    if(freeIndex === -1){ log('Нет свободного слота — купите слот'); return; }
    const g = shopGPUs.find(x=>x.id===gpuId);
    if(!g) return;
    if(state.money < g.price){ log('Недостаточно денег для покупки GPU'); return; }

    // check power capacity after install
    const totals = computeTotals();
    const currentPower = totals.totalPower;
    if(currentPower + g.power > state.powerCapacity){
      log('Недостаточно мощности питания — апгрейдите питание');
      return;
    }

    state.money -= g.price;
    state.slots[freeIndex].gpu = cloneGpu(gpuId);
    state.slots[freeIndex].overclock = false;
    state.slots[freeIndex].brokenUntil = null;
    log(`Куплена ${g.name} за $${g.price} и установлена в слот ${freeIndex+1}`);
    updateUI(); save();
  }

  // open install menu (simple: allow to buy from store then place here)
  function openInstallMenu(slotIndex){
    // show available GPUs to buy via prompt for simplicity
    const choices = shopGPUs.filter(g=>g.id!=='starter').map(g=>`${g.name} — $${g.price} (id:${g.id})`).join('\n');
    const sel = prompt(`Доступные GPU:\n${choices}\n\nВведите id (например: gpu1) или отмена:`);
    if(sel){ buyGpuToSlot(sel.trim(), slotIndex); }
  }

  function buyGpuToSlot(id, slotIndex){
    const g = shopGPUs.find(x=>x.id===id);
    if(!g){ log('Неверный id GPU'); return; }
    if(state.money < g.price){ log('Нет денег'); return; }
    const totals = computeTotals();
    if(totals.totalPower + g.power > state.powerCapacity){ log('Недостаточно мощности'); return; }
    state.money -= g.price;
    state.slots[slotIndex].gpu = cloneGpu(id);
    state.slots[slotIndex].brokenUntil = null;
    state.slots[slotIndex].overclock = false;
    log(`Куплена ${g.name} и установлена в слот ${slotIndex+1}`);
    updateUI(); save();
  }

  // sell GPU at 50%
  function sellGpu(slotIndex){
    const s = state.slots[slotIndex];
    if(!s.gpu){ log('Слот пуст'); return; }
    const price = Math.round(s.gpu.price * 0.5);
    state.money += price;
    log(`Продана ${s.gpu.name} за $${price}`);
    s.gpu = null;
    s.overclock = false;
    s.brokenUntil = null;
    updateUI(); save();
  }

  // toggle overclock: increases hashrate by 25% and power by 15%, increases failure chance; costs small fee to enable once
  function toggleOC(slotIndex){
    const s = state.slots[slotIndex];
    if(!s.gpu){ log('Нет GPU'); return; }
    s.overclock = !s.overclock;
    log(`${s.gpu.name} overclock ${s.overclock ? 'включён' : 'выключен'}`);
    updateUI(); save();
  }

  // repair now for money: cost = price * repairFactor (maybe reduced by maintenance)
  function repairNow(slotIndex){
    const s = state.slots[slotIndex];
    if(!s.gpu){ log('Нет GPU'); return; }
    if(!s.brokenUntil || s.brokenUntil <= now()){ log('GPU не сломана'); return; }
    const baseCost = s.gpu.price * s.gpu.repairFactor;
    const cost = state.maintenanceTeam ? Math.round(baseCost * 0.6) : Math.round(baseCost);
    if(state.money < cost){ log('Недостаточно денег для ремонта'); return; }
    state.money -= cost;
    s.brokenUntil = null;
    log(`${s.gpu.name} отремонтирована за $${cost}`);
    updateUI(); save();
  }

  // logging
  function log(msg){
    const time = new Date().toLocaleTimeString();
    const el = document.createElement('div');
    el.textContent = `[${time}] ${msg}`;
    logEl.prepend(el);
    // keep log small
    while(logEl.children.length > 200) logEl.removeChild(logEl.lastChild);
  }

  // reset
  btnReset.addEventListener('click', ()=>{
    if(!confirm('Сбросить прогресс?')) return;
    localStorage.removeItem(SAVE_KEY);
    location.reload();
  });

  // autosave interval
  setInterval(save, 15000);

  // game tick every second
  setInterval(tick, 1000);

  // initial UI update
  updateUI();

  // Telegram WebApp
  try{
    if(window.Telegram && window.Telegram.WebApp){
      window.Telegram.WebApp.ready();
      try{ window.Telegram.WebApp.expand(); }catch(e){}
    }
  }catch(e){}

  // expose for debug
  window.MINER = { state, shopGPUs, computeTotals, save, load };

  // initial notifications
  log('Старт игры. У вас $' + state.money.toFixed(2));
  log('Первая видеокарта уже установлена в старом компьютере.');
})();

Что добавлено и как играть - Начальный капитал $1000 и одна старая видеокарта в первом слоте (даёт ~1$/с). - Магазин видеокарт: разные уровни с ценой, хешрейтом (H/s) и потреблением (W). - Слоты (ячейки): нельзя поставить карту без свободного слота. - Продажа карты даёт 50% стоимости. - Электропитание (мощность, W): апгрейдить нужно, чтобы ставить больше карт. - Услуги: охлаждение (меньше поломок), сервисная команда (быстрее/дешевле ремонт), Boost (временный +50% хеша). - Overclock для каждой карты (вкл/выкл) — увеличивает хеш и риск. - Электричество стоит $/kWh (можно менять) — влияет на себестоимость/чистую прибыль. - Сломанные карты отключаются; можно чинить вручную или авто‑ремонт сервисной командой. - Игра сохраняется в localStorage.
